cmake_minimum_required(VERSION 3.26)
project(structural_value_publication LANGUAGES CXX)
find_package(hgraph CONFIG REQUIRED)
add_executable(structural_value_publication_native native.cpp)
target_compile_features(structural_value_publication_native PRIVATE cxx_std_23)
target_link_libraries(structural_value_publication_native PRIVATE hgraph::core)
hgraph_link_python_embedding(structural_value_publication_native)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/native.cpp")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/native.cpp" source_hash)
target_compile_definitions(structural_value_publication_native PRIVATE SOURCE_SHA256="${source_hash}")
