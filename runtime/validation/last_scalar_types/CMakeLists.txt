cmake_minimum_required(VERSION 3.26)
project(last_scalar_types LANGUAGES CXX)
find_package(hgraph CONFIG REQUIRED)
add_executable(last_scalar_types_native native.cpp)
target_compile_features(last_scalar_types_native PRIVATE cxx_std_23)
target_link_libraries(last_scalar_types_native PRIVATE hgraph::core)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/native.cpp")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/native.cpp" source_hash)
target_compile_definitions(last_scalar_types_native PRIVATE SOURCE_SHA256="${source_hash}")
