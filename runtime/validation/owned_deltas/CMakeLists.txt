cmake_minimum_required(VERSION 3.26)
project(owned_deltas LANGUAGES CXX)
find_package(hgraph CONFIG REQUIRED)
add_executable(owned_deltas_native native.cpp)
target_compile_features(owned_deltas_native PRIVATE cxx_std_23)
target_link_libraries(owned_deltas_native PRIVATE hgraph::core)
hgraph_link_python_embedding(owned_deltas_native)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/native.cpp")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/native.cpp" source_hash)
target_compile_definitions(owned_deltas_native PRIVATE SOURCE_SHA256="${source_hash}")
