cmake_minimum_required(VERSION 3.26)
project(ordinary_value_sequences LANGUAGES CXX)
find_package(hgraph CONFIG REQUIRED)
add_executable(value_sequences_native native.cpp)
target_compile_features(value_sequences_native PRIVATE cxx_std_23)
target_link_libraries(value_sequences_native PRIVATE hgraph::core)
hgraph_link_python_embedding(value_sequences_native)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/native.cpp" source_hash)
target_compile_definitions(value_sequences_native PRIVATE SOURCE_SHA256="${source_hash}")
