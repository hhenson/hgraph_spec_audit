cmake_minimum_required(VERSION 3.26)
project(delta_eval_native LANGUAGES CXX)
find_package(hgraph CONFIG REQUIRED)
add_executable(delta_eval_native native_scalar.cpp)
target_compile_features(delta_eval_native PRIVATE cxx_std_23)
target_link_libraries(delta_eval_native PRIVATE hgraph::core)
hgraph_link_python_embedding(delta_eval_native)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/native_scalar.cpp" source_hash)
target_compile_definitions(delta_eval_native PRIVATE DELTA_SOURCE_SHA256="${source_hash}")
